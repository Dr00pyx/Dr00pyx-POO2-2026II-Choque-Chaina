def _registrar(self) -> None:
    try:
        cod = int(self._ent_cod.get())
        nom = self._ent_nom.get().strip()
        if not nom:
            raise ValueError("Nombre vacío")
        raw = self._ent_notas.get().strip()
        notas = [float(x) for x in raw.split(",") if x.strip()] if raw else []
        e = Estudiante(cod, nom, 0.0, notas)
        self._svc.registrar(e)
        self._actualizar_lista()
        self._ent_cod.delete(0, "end")
        self._ent_nom.delete(0, "end")
        self._ent_notas.delete(0, "end")
    except ValueError as ex:
        messagebox.showerror("Error", str(ex))

def _buscar(self) -> None:
    raw_cod = self._ent_buscar.get().strip()
    if not raw_cod:
        self._actualizar_lista()
        return
    try:
        cod = int(raw_cod)
        e = self._svc.buscar(cod)
        self._lb.delete(0, "end")
        if e:
            ppa_str = f"{e.ppa:.2f}" if e.notas else "sin notas"
            self._lb.insert("end", f"[{e.codigo:04d}] {e.nombre:<28} PPA: {ppa_str}")
        else:
            messagebox.showinfo("Búsqueda", f"No se encontró código {cod}")
            self._actualizar_lista()
    except ValueError:
        messagebox.showerror("Error", "El código debe ser numérico")

2. Actividad 2A: PyQt5 con Signal/Slot

Patrón Observer nativo, QTableWidget extensible, resaltado en rojo para notas < 10.5 y botón de eliminación reactiva.

pyqt5_app.py

Observer Mediante pyqtSignal

A diferencia de los callbacks convencionales de GUI, Qt implementa el patrón Observer desacoplado:

1. Señal: estudiante_guardado = pyqtSignal(object) y estudiante_eliminado = pyqtSignal(int).

2. Emisión: Ocurre dentro de ModeloAcademico al mutar datos: self.estudiante_guardado.emit(e).

3. Suscripción (Slot): self._mod.estudiante_guardado.connect(self._on_estudiante_guardado).

Formato Condicional: Durante _recargar_tabla(), si e.ppa < 10.5, se invoca ppa_item.setForeground(QColor("#C00000")), pintando automáticamente en color carmesí las calificaciones desaprobadas.

Fragmento: ModeloAcademico y Slot de Recarga
class ModeloAcademico(QObject):
    estudiante_guardado = pyqtSignal(object)
    estudiante_eliminado = pyqtSignal(int)

    def __init__(self, svc) -> None:
        super().__init__()
        self._svc = svc

    def registrar(self, e: Estudiante) -> None:
        self._svc.registrar(e)
        self.estudiante_guardado.emit(e)

    def eliminar(self, cod: int) -> None:
        self._svc.eliminar(cod)
        self.estudiante_eliminado.emit(cod)

def _recargar_tabla(self) -> None:
    self._tbl.setRowCount(0)
    for e in self._mod.listar():
        r = self._tbl.rowCount()
        self._tbl.insertRow(r)
        self._tbl.setItem(r, 0, QTableWidgetItem(str(e.codigo)))
        self._tbl.setItem(r, 1, QTableWidgetItem(e.nombre))
        ppa_item = QTableWidgetItem(f"{e.ppa:.2f}")
        if e.ppa < 10.5:
            ppa_item.setForeground(QColor("#C00000"))
        self._tbl.setItem(r, 2, ppa_item)
        self._tbl.setItem(r, 3, QTableWidgetItem(str(e.notas)))

3. Verificación de la Separación MVC

Garantía arquitectónica: la Vista nunca toca el repositorio directamente.

V

Capa de Vista

Representada por AppEstudiantes y VistaEstudiantes. Solo captura eventos de interfaz, valida entradas básicas de formulario y renderiza los datos suministrados por el modelo.

C

Capa de Controlador / Modelo

ModeloAcademico actúa como envoltorio reactivo de ServicioEstudiante. Aplica la estrategia de cálculo de notas (Strategy) y notifica los cambios mediante señales.

M

Capa de Persistencia

RepositorioJSON encapsula la lectura y escritura del archivo datos.json. Totalmente aislado de cualquier biblioteca gráfica (`tkinter` o `PyQt5`).
